def select(name, original, *, automatic):
    """Change only selected component switches in an original validated workload."""
    if type(automatic) is not bool:
        raise ValueError('automatic selection must be an explicit settled boolean')
    campaign.validate_paired_protocol(name, original)
    result = copy.deepcopy(original)
    reference = result['arms']['reference']['env']
    if AUTO in reference or reference[NGRAM] != '0' or reference[VISION] != '0' or reference[TAIL] != '0':
        raise ValueError('original explicit reference controls changed')
    reference[AUTO] = '0'
    combined = result['arms']['combined']['env']
    if name == 'actual-default-one-token':
        if combined != {}:
            raise ValueError('actual deployment defaults must have an empty environment')
    else:
        if AUTO in combined or combined[NGRAM] != '0' or combined[VISION] != '0' or combined[TAIL] != '0':
            raise ValueError('original explicit candidate controls changed')
        combined[NGRAM] = '1'
        combined[VISION] = '256'
        # Original templates deliberately include manual scope controls. They
        # suppress inherited automatic selection, so make selection explicit.
        combined[AUTO] = '1' if automatic else '0'
    restored = restore(name, result)
    if restored != original:
        raise ValueError('selection changed a workload or acceptance field')
    return result


def restore(name, selected):
    result = copy.deepcopy(selected)
    result['arms']['reference']['env'].pop(AUTO)
    if name != 'actual-default-one-token':
        combined = result['arms']['combined']['env']
        combined.pop(AUTO)
        combined[NGRAM] = '0'
        combined[VISION] = '0'
    return result


def validate(name, selected, original, *, automatic):
    if selected != select(name, original, automatic=automatic):
        raise ValueError('final component controls or original workload changed')
    campaign.validate_paired_protocol(name, restore(name, selected))
    return True
