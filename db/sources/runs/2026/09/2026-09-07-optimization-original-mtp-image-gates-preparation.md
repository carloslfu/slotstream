def execute(command, target, seconds, policy, assessment, limit, attempt, progress_path=None):
    target.mkdir(parents=True, exist_ok=False)
    c.write_new(target / 'attempt.json', attempt)
    result = {'passed': False, 'completed': False, 'qualified': False, 'attempt_sha256': digest(target / 'attempt.json')}
    saved = {k: v for k, v in os.environ.items() if k.startswith(('SLOTSTREAM_', 'SS_DEBUG'))}
    start = time.monotonic()
    stop = threading.Event()
    def interrupted(number, _):
        raise KeyboardInterrupt(f'owned optional qualification interrupted by signal {number}')
    handlers = {n: signal.signal(n, interrupted) for n in (signal.SIGINT, signal.SIGTERM)}
    def progress():
        seen = 0
        while not stop.wait(10):
            try:
                rows = [json.loads(line) for line in progress_path.read_text().splitlines()]
                if len(rows) > seen:
                    seen = len(rows)
                    print(json.dumps({'completed_cells': seen, 'fixed_total_cells': 32,
                                      'eligible_cells': sum(r.get('valid') is True for r in rows)}), flush=True)
            except (OSError, json.JSONDecodeError):
                pass
    observer = threading.Thread(target=progress, daemon=True) if progress_path else None
    try:
        for key in saved:
            os.environ.pop(key)
        if observer:
            observer.start()
        def snapshot():
            state = vm_snapshot()
            require_normal(pressure_snapshot())
            return state
        with (target / 'stdout.txt').open('x') as out, (target / 'stderr.txt').open('x') as err:
            child = guarded_run(command, cwd=ROOT, stdout=out, stderr=err,
                                record_path=target / 'memory.json', snapshot=snapshot,
                                policy=policy, classification=attempt['classification'])
        result['exit_code'] = child.returncode
        result.update(assessment(child.returncode))
    except BaseException as error:
        result['error'] = f'{type(error).__name__}: {error}'
    finally:
        stop.set()
        if observer and observer.is_alive():
            observer.join(timeout=1)
        for n, handler in handlers.items():
            signal.signal(n, handler)
        for key in list(os.environ):
            if key.startswith(('SLOTSTREAM_', 'SS_DEBUG')):
                os.environ.pop(key)
        os.environ.update(saved)
        result['elapsed_seconds'] = time.monotonic() - start
        try:
            load()
            result['proofs_unchanged'] = True
            result['after'] = preflight(0)
            result['remaining_jobs'] = competing_jobs()
            result['cleanup_complete'] = not result['remaining_jobs']
        except BaseException as error:
            result['cleanup_error'] = str(error)
        result['within_reservation'] = time.time() <= limit
        valid = result.get('proofs_unchanged') and result.get('cleanup_complete') and result['within_reservation'] and not result.get('cleanup_error')
        for key in ('passed', 'completed', 'qualified'):
            result[key] = bool(result[key] and valid)
        result['stdout_sha256'] = digest(target / 'stdout.txt') if (target / 'stdout.txt').exists() else None
        c.write_new(target / 'receipt.json', result)
    return result
